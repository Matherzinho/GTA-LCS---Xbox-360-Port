cmake_minimum_required(VERSION 3.20)
project(GTA_LCS_XSDK LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LCS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../work/lcs/relcs" CACHE PATH "Pinned reLCS checkout")
# Use portable native texture storage; RW_XSDK selects the custom Xenos device.
set(LIBRW_PLATFORM NULL CACHE STRING "" FORCE)
set(LIBRW_TOOLS OFF CACHE BOOL "" FORCE)
set(LIBRW_INSTALL OFF CACHE BOOL "" FORCE)
set(LIBRW_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LCS_SOURCE}/vendor/librw" "${CMAKE_BINARY_DIR}/librw")
file(GLOB_RECURSE LCS_SOURCES CONFIGURE_DEPENDS "${LCS_SOURCE}/src/*.cpp" "${LCS_SOURCE}/src/*.c")
list(FILTER LCS_SOURCES EXCLUDE REGEX "/skel/(win|glfw)/|/audio/(oal|eax)/|/audio/sampman_|/extras/(debugmenu|imgui)|/extras/shaders/|/control/ScriptDebug.cpp$")
list(FILTER LCS_SOURCES EXCLUDE REGEX "/core/(CdStreamPosix|FileMgr).cpp$|/skel/crossplatform.cpp$|/text/Text.cpp$")
file(GLOB_RECURSE LCS_HEADERS CONFIGURE_DEPENDS "${LCS_SOURCE}/src/*.h")
set(LCS_INCLUDES "${LCS_SOURCE}/src" "${LCS_SOURCE}/vendor/librw" "${CMAKE_CURRENT_SOURCE_DIR}/platform")
foreach(header IN LISTS LCS_HEADERS)
 get_filename_component(directory "${header}" DIRECTORY)
 list(APPEND LCS_INCLUDES "${directory}")
endforeach()
list(REMOVE_DUPLICATES LCS_INCLUDES)
add_library(lcs_gameplay STATIC ${LCS_SOURCES} platform/xsdk_input.cpp)
add_library(lcs_fileio STATIC "${LCS_SOURCE}/src/core/FileMgr.cpp" "${LCS_SOURCE}/src/skel/crossplatform.cpp" platform/xsdk_paths.cpp platform/xsdk_cdstream.cpp)
target_include_directories(lcs_fileio PRIVATE ${LCS_INCLUDES} "${CMAKE_BINARY_DIR}/librw/src")
target_compile_definitions(lcs_fileio PRIVATE LIBRW RW_NULL RWDEVICE=xsdk GTA_LCS_XSDK NDEBUG MASTER BIGENDIAN USE_UNNAMED_SEM)
target_compile_options(lcs_fileio PRIVATE -fpermissive -Wno-unknown-pragmas -Wno-narrowing)
target_link_libraries(lcs_fileio PUBLIC librw)
add_library(lcs_text STATIC "${LCS_SOURCE}/src/text/Text.cpp")
target_include_directories(lcs_text PRIVATE ${LCS_INCLUDES} "${CMAKE_BINARY_DIR}/librw/src")
target_compile_definitions(lcs_text PRIVATE LIBRW RW_NULL RWDEVICE=xsdk GTA_LCS_XSDK NDEBUG MASTER BIGENDIAN)
target_compile_options(lcs_text PRIVATE -fpermissive -Wno-unknown-pragmas -Wno-narrowing -ffunction-sections -fdata-sections)
target_link_libraries(lcs_text PUBLIC lcs_fileio)
target_include_directories(lcs_gameplay PRIVATE ${LCS_INCLUDES} "${CMAKE_BINARY_DIR}/librw/src")
target_compile_definitions(lcs_gameplay PRIVATE LIBRW RW_NULL RWDEVICE=xsdk GTA_LCS_XSDK NDEBUG MASTER BIGENDIAN USE_UNNAMED_SEM)
target_compile_options(lcs_gameplay PRIVATE -fpermissive -Wno-unknown-pragmas -Wno-narrowing)
target_link_libraries(lcs_gameplay PUBLIC librw lcs_fileio lcs_text)
target_compile_definitions(librw PUBLIC BIGENDIAN)
target_compile_definitions(librw PRIVATE RW_XSDK RWDEVICE=xsdk)
target_compile_options(librw PRIVATE -mcpu=powerpc -mtune=cell -ffp-contract=off)
target_sources(librw PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/platform/rw_xsdk_backend.cpp")
target_include_directories(librw PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/platform" "${LCS_SOURCE}/vendor/librw")
target_link_libraries(librw PUBLIC lcs_gpu)

add_executable(lcs_data_probe EXCLUDE_FROM_ALL platform/data_probe.c)
target_include_directories(lcs_data_probe PRIVATE platform)
target_compile_options(lcs_data_probe PRIVATE -O2 -mno-powerpc64 -ffp-contract=off)
xsdk_add_xex(lcs_data_probe TITLE_ID 0x4C435350 NAME "LCS XSDK data probe - not game")
# Native Xenos device, reused from the user's XSDK AvP bridge.
# Connected to librw 2D and basic 3D. Lighting/skin/material effects are pending.
add_library(lcs_gpu STATIC platform/gpu/xdev.c)
target_include_directories(lcs_gpu PUBLIC platform/gpu)
target_compile_options(lcs_gpu PRIVATE -O2 -mcpu=powerpc -mtune=cell -ffp-contract=off)
xsdk_add_shaders(lcs_gpu platform/gpu/tl.vert platform/gpu/tl.frag platform/gpu/fxaa.frag)

add_executable(lcs_render_probe EXCLUDE_FROM_ALL platform/render_probe.cpp)
target_include_directories(lcs_render_probe PRIVATE platform)
target_link_libraries(lcs_render_probe PRIVATE librw)
target_compile_options(lcs_render_probe PRIVATE -mcpu=powerpc -mtune=cell -ffp-contract=off)
xsdk_add_xex(lcs_render_probe TITLE_ID 0x4C435352 NAME "LCS XSDK render diagnostic - not game")

add_executable(lcs_model_probe EXCLUDE_FROM_ALL platform/model_probe.cpp)
target_include_directories(lcs_model_probe PRIVATE platform)
target_link_libraries(lcs_model_probe PRIVATE librw)
target_compile_options(lcs_model_probe PRIVATE -mcpu=powerpc -mtune=cell -ffp-contract=off)
xsdk_add_xex(lcs_model_probe TITLE_ID 0x4C43534D NAME "LCS XSDK model diagnostic - not game")

add_executable(lcs_io_probe EXCLUDE_FROM_ALL platform/io_probe.cpp)
target_include_directories(lcs_io_probe PRIVATE ${LCS_INCLUDES} "${CMAKE_BINARY_DIR}/librw/src")
target_compile_definitions(lcs_io_probe PRIVATE LIBRW RW_NULL RWDEVICE=xsdk GTA_LCS_XSDK NDEBUG MASTER BIGENDIAN)
target_compile_options(lcs_io_probe PRIVATE -fpermissive -Wno-unknown-pragmas -Wno-narrowing)
target_link_libraries(lcs_io_probe PRIVATE lcs_fileio)
xsdk_add_xex(lcs_io_probe TITLE_ID 0x4C435349 NAME "LCS XSDK IO diagnostic - not game")

add_executable(lcs_text_probe EXCLUDE_FROM_ALL platform/text_probe.cpp)
target_include_directories(lcs_text_probe PRIVATE ${LCS_INCLUDES} "${CMAKE_BINARY_DIR}/librw/src")
target_compile_definitions(lcs_text_probe PRIVATE LIBRW RW_NULL RWDEVICE=xsdk GTA_LCS_XSDK NDEBUG MASTER BIGENDIAN)
target_compile_options(lcs_text_probe PRIVATE -fpermissive -Wno-unknown-pragmas -Wno-narrowing)
target_link_options(lcs_text_probe PRIVATE -Wl,--gc-sections)
target_link_libraries(lcs_text_probe PRIVATE lcs_text)
xsdk_add_xex(lcs_text_probe TITLE_ID 0x4C435354 NAME "LCS XSDK English text diagnostic - not game")

# Actual game startup path, with the upstream silent sample backend until native
# audio is ready. This target is internal and is not a console release.
add_executable(lcs_boot EXCLUDE_FROM_ALL platform/xsdk_boot.cpp platform/xsdk_video_test.cpp platform/xsdk_stress_test.cpp platform/xsdk_gameplay_test.cpp platform/xsdk_safehouse_test.cpp platform/xsdk_platform.cpp platform/xsdk_pipelines.cpp platform/xsdk_sampman.cpp platform/xsdk_sound_sink.c)
target_include_directories(lcs_boot PRIVATE ${LCS_INCLUDES} "${CMAKE_BINARY_DIR}/librw/src")

target_compile_options(lcs_boot PRIVATE -fpermissive -Wno-unknown-pragmas -Wno-narrowing -ffunction-sections -fdata-sections)
target_link_options(lcs_boot PRIVATE -Wl,--gc-sections "-Wl,-Map=${CMAKE_BINARY_DIR}/lcs_boot.map")
target_link_libraries(lcs_boot PRIVATE -Wl,--start-group lcs_gameplay lcs_text lcs_fileio librw -Wl,--end-group)
xsdk_add_xex(lcs_boot TITLE_ID 0x4C435342 NAME "LCS XSDK native audio and gameplay bring-up")

add_executable(lcs_scene_probe EXCLUDE_FROM_ALL platform/scene_probe.cpp)
target_include_directories(lcs_scene_probe PRIVATE platform)
target_link_libraries(lcs_scene_probe PRIVATE librw)
